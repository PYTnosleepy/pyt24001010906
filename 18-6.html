    USER ||--o{ SESSION : "发起"

    SESSION ||--o{ TICKET : "转人工"

    USER {
        bigint id PK "用户 ID"
        varchar nickname "昵称"
        varchar channel "来源渠道"
        datetime created_at "注册时间"
    }

    SESSION {
        bigint id PK "会话 ID"
        bigint user_id FK "所属用户"
        varchar status "会话状态"
        datetime started_at "开始时间"
    }

    MESSAGE {
        bigint id PK "消息 ID"
        bigint session_id FK "所属会话"
        bigint intent_id FK "命中意图"
        varchar role "角色：用户 / 机器人"
        text content "消息内容"
        datetime sent_at "发送时间"
    }

    INTENT {
        bigint id PK "意图 ID"
        varchar name "意图名称"
        decimal confidence "置信度"
    }

    KB_ITEM {
        bigint id PK "知识条目 ID"
        bigint intent_id FK "关联意图"
        varchar title "标题"
        varchar category "分类"
        varchar vector_id "向量索引 ID"
    }

    QA_PAIR {
        bigint id PK "问答对 ID"
        bigint kb_id FK "所属知识条目"
        text question "标准问题"
        text answer "标准答案"
    }

    TICKET {
        bigint id PK "工单 ID"
        bigint session_id FK "来源会话"
        bigint agent_id FK "处理坐席"
        varchar status "工单状态"
        datetime created_at "创建时间"
    }

    AGENT {
        bigint id PK "坐席 ID"
        varchar name "坐席姓名"
        varchar group_name "所属班组"
    }
